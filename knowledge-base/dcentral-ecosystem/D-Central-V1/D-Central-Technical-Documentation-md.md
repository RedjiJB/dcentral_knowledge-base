### Routing Optimization

**Energy-Aware Routing Algorithm**:

```python
class EnergyAwareRouter:
    def __init__(self):
        self.node_energy_states = {}
        self.routing_table = {}
        self.ml_predictor = EnergyPredictor()
    
    def calculate_route_cost(self, path, packet_size):
        total_cost = 0
        for hop in path:
            node_energy = self.node_energy_states[hop.node_id]
            
            # Base transmission cost
            transmission_cost = packet_size * hop.energy_per_bit
            
            # Energy state multiplier
            if node_energy.source == "battery":
                energy_multiplier = 2.0  # Prefer avoiding battery nodes
            elif node_energy.source == "solar":
                solar_availability = self.get_solar_prediction(hop.location)
                energy_multiplier = 1.0 if solar_availability > 0.7 else 1.5
            else:  # mains power
                energy_multiplier = 0.8  # Prefer mains-powered nodes
            
            # Remaining energy factor
            energy_factor = max(0.5, node_energy.remaining_pct)
            
            hop_cost = transmission_cost * energy_multiplier / energy_factor
            total_cost += hop_cost
        
        return total_cost
    
    def find_optimal_path(self, source, destination, packet_size):
        # Dijkstra's algorithm with energy-aware cost function
        unvisited = set(self.get_all_nodes())
        distances = {node: float('inf') for node in unvisited}
        distances[source] = 0
        previous = {}
        
        while unvisited:
            current = min(unvisited, key=lambda node: distances[node])
            unvisited.remove(current)
            
            if current == destination:
                break
            
            for neighbor in self.get_neighbors(current):
                if neighbor in unvisited:
                    path_segment = [current, neighbor]
                    alt_distance = distances[current] + self.calculate_route_cost(path_segment, packet_size)
                    
                    if alt_distance < distances[neighbor]:
                        distances[neighbor] = alt_distance
                        previous[neighbor] = current
        
        # Reconstruct path
        path = []
        current = destination
        while current in previous:
            path.insert(0, current)
            current = previous[current]
        path.insert(0, source)
        
        return path
```

```go
type BorderGateway struct {
    MeshInterface *MeshInterface
    InternetUplinks []*InternetUplink
    LoadBalancer *LoadBalancer
    SecurityGateway *SecurityGateway
    QoSManager *QoSManager
}

type InternetUplink struct {
    Provider string
    Interface string
    Bandwidth int64
    Latency time.Duration
    Cost float64
    Reliability float64
}

func (bg *BorderGateway) RouteToInternet(packet *Packet) error {
    // Apply security filtering
    if !bg.SecurityGateway.ValidatePacket(packet) {
        return errors.New("packet failed security validation")
    }
    
    // Select optimal uplink
    uplink := bg.LoadBalancer.SelectUplink(packet.Destination, packet.Priority)
    
    // Apply QoS policies
    err := bg.QoSManager.ApplyTrafficShaping(packet, uplink)
    if err != nil {
        return fmt.Errorf("QoS application failed: %w", err)
    }
    
    // Forward packet
    return uplink.ForwardPacket(packet)
}

func (lb *LoadBalancer) SelectUplink(destination string, priority int) *InternetUplink {
    // Weighted selection based on performance and cost
    candidates := lb.getAvailableUplinks()
    weights := make([]float64, len(candidates))
    
    for i, uplink := range candidates {
        // Calculate weight based on multiple factors
        latencyWeight := 1.0 / float64(uplink.Latency.Milliseconds())
        bandwidthWeight := float64(uplink.Bandwidth) / 1000000 // Normalize to Mbps
        reliabilityWeight := uplink.Reliability
        costWeight := 1.0 / uplink.Cost // Lower cost = higher weight
        
        weights[i] = (latencyWeight * 0.3) + (bandwidthWeight * 0.3) + 
                    (reliabilityWeight * 0.3) + (costWeight * 0.1)
        
        // Priority adjustment
        if priority >= 8 { // High priority traffic
            weights[i] *= 1.5
        }
    }
    
    // Weighted random selection
    return lb.weightedRandomSelect(candidates, weights)
}
```

```go
type MeshScheduler struct {
    nodes map[NodeID]*NodeInfo
    workloads map[WorkloadID]*WorkloadSpec
    optimizer *SchedulingOptimizer
    monitor *ResourceMonitor
}

type NodeInfo struct {
    ID NodeID
    Resources *ResourceCapacity
    EnergyState *EnergyState
    NetworkPosition *NetworkPosition
    Capabilities []string
}

type WorkloadSpec struct {
    ID WorkloadID
    ResourceRequirements *ResourceRequirements
    Constraints []Constraint
    Priority int
    SLA *SLARequirements
}

func (ms *MeshScheduler) ScheduleWorkload(workload *WorkloadSpec) (*SchedulingDecision, error) {
    // Filter nodes based on constraints
    candidateNodes := ms.filterNodesByConstraints(workload.Constraints)
    
    // Score nodes based on multiple factors
    nodeScores := make(map[NodeID]float64)
    for _, node := range candidateNodes {
        score := ms.calculateNodeScore(node, workload)
        nodeScores[node.ID] = score
    }
    
    // Select optimal node
    optimalNode := ms.selectOptimalNode(nodeScores, workload.Priority)
    
    // Create scheduling decision
    decision := &SchedulingDecision{
        WorkloadID: workload.ID,
        TargetNode: optimalNode,
        Resources: workload.ResourceRequirements,
        PlacementScore: nodeScores[optimalNode],
        ScheduledAt: time.Now(),
    }
    
    return decision, nil
}

func (ms *MeshScheduler) calculateNodeScore(node *NodeInfo, workload *WorkloadSpec) float64 {
    // Resource availability score (0-1)
    resourceScore := ms.calculateResourceScore(node.Resources, workload.ResourceRequirements)
    
    // Energy efficiency score (0-1)
    energyScore := ms.calculateEnergyScore(node.EnergyState, workload.ResourceRequirements)
    
    // Network position score (0-1)
    networkScore := ms.calculateNetworkScore(node.NetworkPosition, workload.SLA)
    
    // Capability match score (0-1)
    capabilityScore := ms.calculateCapabilityScore(node.Capabilities, workload.Constraints)
    
    // Weighted composite score
    return (resourceScore * 0.3) + (energyScore * 0.2) + 
           (networkScore * 0.3) + (capabilityScore * 0.2)
}
```

```typescript
interface CRDTOperationLog {
    nodeId: string;
    timestamp: number;
    operation: CRDTOperation;
    vectorClock: VectorClock;
}

class DistributedDataFabric {
    private nodes: Map<string, NodeConnection>;
    private operationLog: CRDTOperationLog[];
    private conflictResolver: ConflictResolver;
    private replicationManager: ReplicationManager;

    async writeData(key: string, value: any, consistency: ConsistencyLevel): Promise<WriteResult> {
        const operation: CRDTOperation = {
            type: 'SET',
            key,
            value,
            nodeId: this.getNodeId(),
            timestamp: Date.now(),
            vectorClock: this.generateVectorClock()
        };

        // Apply operation locally
        await this.applyLocalOperation(operation);

        // Determine replication strategy based on consistency level
        const replicationStrategy = this.getReplicationStrategy(consistency);
        
        // Replicate to other nodes
        const replicationResults = await this.replicationManager.replicate(
            operation, 
            replicationStrategy
        );

        return {
            success: true,
            localTimestamp: operation.timestamp,
            replicationResults,
            consistencyAchieved: this.verifyConsistency(replicationResults, consistency)
        };
    }

    async readData(key: string, consistency: ConsistencyLevel): Promise<ReadResult> {
        switch (consistency) {
            case ConsistencyLevel.EVENTUAL:
                return this.readLocal(key);
            
            case ConsistencyLevel.STRONG:
                return this.readWithQuorum(key);
            
            case ConsistencyLevel.CAUSAL:
                return this.readWithCausalConsistency(key);
            
            default:
                throw new Error(`Unsupported consistency level: ${consistency}`);
        }
    }

    private async resolveConflicts(operations: CRDTOperation[]): Promise<any> {
        // Sort operations by vector clock for causal ordering
        const sortedOps = operations.sort((a, b) => 
            this.compareVectorClocks(a.vectorClock, b.vectorClock)
        );

        let resolvedValue = null;
        for (const op of sortedOps) {
            resolvedValue = this.conflictResolver.resolve(resolvedValue, op);
        }

        return resolvedValue;
    }
}

class ReplicationManager {
    async replicate(operation: CRDTOperation, strategy: ReplicationStrategy): Promise<ReplicationResult[]> {
        const targetNodes = this.selectReplicationTargets(strategy);
        const replicationPromises = targetNodes.map(node => 
            this.replicateToNode(node, operation)
        );

        const results = await Promise.allSettled(replicationPromises);
        return results.map((result, index) => ({
            nodeId: targetNodes[index].id,
            success: result.status === 'fulfilled',
            error: result.status === 'rejected' ? result.reason : null,
            latency: this.measureReplicationLatency(targetNodes[index])
        }));
    }

    private selectReplicationTargets(strategy: ReplicationStrategy): NodeConnection[] {
        switch (strategy.type) {
            case 'GEOGRAPHIC':
                return this.selectGeographicallyDistributedNodes(strategy.replicas);
            
            case 'ENERGY_AWARE':
                return this.selectEnergyEfficientNodes(strategy.replicas);
            
            case 'NETWORK_AWARE':
                return this.selectNetworkOptimalNodes(strategy.replicas);
            
            default:
                return this.selectRandomNodes(strategy.replicas);
        }
    }
}
```

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Dict, List, Any, Optional
import asyncio

@dataclass
class AgentMessage:
    sender_id: str
    receiver_id: str
    message_type: str
    content: Dict[str, Any]
    conversation_id: str
    timestamp: float
    priority: int = 5  # 1-10 scale

class Agent(ABC):
    def __init__(self, agent_id: str, capabilities: List[str]):
        self.agent_id = agent_id
        self.capabilities = capabilities
        self.message_queue = asyncio.Queue()
        self.conversation_contexts = {}
        self.reputation_score = 0.5
    
    @abstractmethod
    async def process_message(self, message: AgentMessage) -> Optional[AgentMessage]:
        pass
    
    async def send_message(self, receiver_id: str, message_type: str, content: Dict[str, Any], 
                          conversation_id: str = None) -> str:
        message = AgentMessage(
            sender_id=self.agent_id,
            receiver_id=receiver_id,
            message_type=message_type,
            content=content,
            conversation_id=conversation_id or self.generate_conversation_id(),
            timestamp=time.time()
        )
        
        # Route message through agent coordinator
        await AgentCoordinator.instance().route_message(message)
        return message.conversation_id

class ServiceBrokerAgent(Agent):

        scored_candidates.sort(key=lambda x: x[1], reverse=True)
        top_candidates = scored_candidates[:5]  # Top 5 candidates

        negotiation_results = []
        for candidate, score in top_candidates:
```

```python
class FederatedLearningCoordinator:

        selected_nodes = self.select_training_nodes()

        model_distribution_tasks = []

            private_updates.append(private_update)

        aggregated_update = await self.secure_aggregate_updates(private_updates)

        evaluation_result = await self.evaluate_global_model()

            round_number=self.round_number,

        # Filter nodes based on availability and resources
        available_nodes = []
        for node_id in candidate_nodes:
            node_info = self.participant_nodes[node_id]
            if (node_info.is_online and 
                node_info.cpu_usage < 0.8 and 
                node_info.has_training_data):
                available_nodes.append(node_id)
        
        # Score nodes based on data quality and network conditions
        node_scores = {}
        for node_id in available_nodes:
            node_info = self.participant_nodes[node_id]
            
            # Data quality score
            data_quality = node_info.data_quality_score
            
            # Network condition score
            network_score = 1.0 / (1.0 + node_info.latency_ms / 1000.0)
            
            # Energy efficiency score
            energy_score = 1.0 if node_info.energy_source == "mains" else 0.7
            
            # Historical reliability score
            reliability_score = node_info.training_completion_rate
            
            # Composite score
            composite_score = (data_quality * 0.4 + 
                             network_score * 0.2 + 
                             energy_score * 0.2 + 
                             reliability_score * 0.2)
            
            node_scores[node_id] = composite_score
        
        # Select top nodes up to maximum participants per round
        sorted_nodes = sorted(node_scores.items(), key=lambda x: x[1], reverse=True)
        max_participants = min(len(sorted_nodes), self.max_participants_per_round)
        
        return [node_id for node_id, score in sorted_nodes[:max_participants]]
```

```solidity
        // Execute transfer
        _transfer(from, to, netAmount);

        uint256 totalVolume = 0;
        // This would typically use a more sophisticated tracking mechanism
        // For brevity, showing simplified calculation

        return (totalVolume * 365 days) / (averageSupply * timeElapsed);
```

```go
package security

import (
    "crypto/rand"
    "errors"
    
    "github.com/cloudflare/circl/kem/kyber/kyber768"
    "github.com/cloudflare/circl/sign/dilithium/mode3"
)

type PostQuantumCrypto struct {
    kemScheme    kyber768.Scheme
    signScheme   mode3.Scheme
    privateKey   mode3.PrivateKey
    publicKey    mode3.PublicKey
    kemPublicKey kyber768.PublicKey
    kemPrivateKey kyber768.PrivateKey
}

func NewPostQuantumCrypto() (*PostQuantumCrypto, error) {
    pqc := &PostQuantumCrypto{
        kemScheme:  kyber768.Scheme{},
        signScheme: mode3.Scheme{},
    }
    
    // Generate signing keys
    publicKey, privateKey, err := pqc.signScheme.GenerateKey(rand.Reader)
    if err != nil {
        return nil, err
    }
    pqc.publicKey = publicKey.(mode3.PublicKey)
    pqc.privateKey = privateKey.(mode3.PrivateKey)
    
    // Generate KEM keys
    kemPublicKey, kemPrivateKey, err := pqc.kemScheme.GenerateKeyPair(rand.Reader)
    if err != nil {
        return nil, err
    }
    pqc.kemPublicKey = kemPublicKey.(kyber768.PublicKey)
    pqc.kemPrivateKey = kemPrivateKey.(kyber768.PrivateKey)
    
    return pqc, nil
}

func (pqc *PostQuantumCrypto) Sign(message []byte) ([]byte, error) {
    return pqc.signScheme.Sign(pqc.privateKey, message, rand.Reader), nil
}

func (pqc *PostQuantumCrypto) Verify(message, signature []byte, publicKey mode3.PublicKey) bool {
    return pqc.signScheme.Verify(publicKey, message, signature)
}

func (pqc *PostQuantumCrypto) Encapsulate() ([]byte, []byte, error) {
    sharedSecret, ciphertext := pqc.kemScheme.Encapsulate(pqc.kemPublicKey, rand.Reader)
    return sharedSecret, ciphertext, nil
}

func (pqc *PostQuantumCrypto) Decapsulate(ciphertext []byte) ([]byte, error) {
    sharedSecret := pqc.kemScheme.Decapsulate(pqc.kemPrivateKey, ciphertext)
    return sharedSecret, nil
}

// Hybrid encryption combining classical and post-quantum cryptography
type HybridEncryption struct {
    postQuantum *PostQuantumCrypto
    classical   *ClassicalCrypto // RSA/ECDSA implementation
}

func (he *HybridEncryption) HybridEncrypt(plaintext []byte, recipientPQPublicKey mode3.PublicKey) (*HybridCiphertext, error) {
    // Generate ephemeral keys for both schemes
    pqSharedSecret, pqCiphertext, err := he.postQuantum.Encapsulate()
    if err != nil {
        return nil, err
    }
    
    classicalSharedSecret, classicalCiphertext, err := he.classical.Encapsulate()
    if err != nil {
        return nil, err
    }
    
    // Combine shared secrets
    combinedSecret := he.combineSecrets(pqSharedSecret, classicalSharedSecret)
    
    // Encrypt plaintext with combined secret
    encryptedData, err := he.symmetricEncrypt(plaintext, combinedSecret)
    if err != nil {
        return nil, err
    }
    
    return &HybridCiphertext{
        PQCiphertext:        pqCiphertext,
        ClassicalCiphertext: classicalCiphertext,
        EncryptedData:       encryptedData,
    }, nil
}
```

### Zero-Trust Network Access

**Network Access Control Implementation**:

```typescript
interface AccessPolicy {
    resourceId: string;
    principal: Principal;
    permissions: Permission[];
    conditions: AccessCondition[];
    expiresAt?: Date;
}

interface Principal {
    type: 'user' | 'device' | 'service';
    identifier: string;
    did: string;
    trustLevel: number; // 0-100
}

interface Permission {
    action: string; // read, write, execute, admin
    scope: string; // specific resource scope
    constraints: string[]; // additional constraints
}

class ZeroTrustAccessController {
    private policyEngine: PolicyEngine;
    private identityVerifier: IdentityVerifier;
    private riskAssessment: RiskAssessmentEngine;
    private auditLogger: AuditLogger;

    async evaluateAccess(
        request: AccessRequest,
        context: SecurityContext
    ): Promise<AccessDecision> {
        // Step 1: Verify identity
        const identityVerification = await this.identityVerifier.verify(
            request.principal.did,
            request.credentials
        );

        if (!identityVerification.isValid) {
            return this.denyAccess(request, 'IDENTITY_VERIFICATION_FAILED');
        }

        // Step 2: Assess risk
        const riskScore = await this.riskAssessment.calculateRisk(
            request,
            context,
            identityVerification.trustLevel
        );

        // Step 3: Apply policies
        const applicablePolicies = await this.policyEngine.findApplicablePolicies(
            request.resourceId,
            request.principal
        );

        let finalDecision = AccessDecisionType.DENY;
        const policyResults: PolicyEvaluationResult[] = [];

        for (const policy of applicablePolicies) {
            const policyResult = await this.evaluatePolicy(
                policy,
                request,
                context,
                riskScore
            );
            policyResults.push(policyResult);

            if (policyResult.decision === AccessDecisionType.ALLOW) {
                finalDecision = AccessDecisionType.ALLOW;
            }
        }

        // Step 4: Apply risk-based additional controls
        let additionalControls: AdditionalControl[] = [];
        if (riskScore > 0.7 && finalDecision === AccessDecisionType.ALLOW) {
            additionalControls = await this.generateAdditionalControls(riskScore);
        }

        const decision: AccessDecision = {
            decision: finalDecision,
            principal: request.principal,
            resourceId: request.resourceId,
            requestedPermissions: request.permissions,
            grantedPermissions: this.calculateGrantedPermissions(policyResults),
            additionalControls,
            riskScore,
            evaluatedAt: new Date(),
            expiresAt: this.calculateExpiration(riskScore),
            auditTrail: this.generateAuditTrail(request, policyResults, riskScore)
        };

        // Step 5: Log decision
        await this.auditLogger.logAccessDecision(decision);

        return decision;
    }

    private async evaluatePolicy(
        policy: AccessPolicy,
        request: AccessRequest,
        context: SecurityContext,
        riskScore: number
    ): Promise<PolicyEvaluationResult> {
        // Evaluate conditions
        const conditionResults = await Promise.all(
            policy.conditions.map(condition => 
                this.evaluateCondition(condition, request, context, riskScore)
            )
        );

        const allConditionsMet = conditionResults.every(result => result.satisfied);

        // Check permission alignment
        const hasRequiredPermissions = this.checkPermissionAlignment(
            policy.permissions,
            request.permissions
        );

        // Check expiration
        const isNotExpired = !policy.expiresAt || policy.expiresAt > new Date();

        const decision = allConditionsMet && hasRequiredPermissions && isNotExpired
            ? AccessDecisionType.ALLOW
            : AccessDecisionType.DENY;

        return {
            policyId: policy.resourceId,
            decision,
            conditionResults,
            hasRequiredPermissions,
            isNotExpired,
            evaluatedAt: new Date()
        };
    }

    private async generateAdditionalControls(riskScore: number): Promise<AdditionalControl[]> {
        const controls: AdditionalControl[] = [];

        if (riskScore > 0.8) {
            controls.push({
                type: 'MULTI_FACTOR_AUTHENTICATION',
                requirement: 'Biometric verification required',
                expiresAt: new Date(Date.now() + 15 * 60 * 1000) // 15 minutes
            });
        }

        if (riskScore > 0.9) {
            controls.push({
                type: 'APPROVAL_REQUIRED',
                requirement: 'Administrator approval required for high-risk access',
                expiresAt: new Date(Date.now() + 60 * 60 * 1000) // 1 hour
            });
        }

        if (riskScore > 0.7) {
            controls.push({
                type: 'ENHANCED_MONITORING',
                requirement: 'All actions will be monitored and logged',
                expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000) // 24 hours
            });
        }

        return controls;
    }
}

class RiskAssessmentEngine {
    async calculateRisk(
        request: AccessRequest,
        context: SecurityContext,
        trustLevel: number
    ): Promise<number> {
        let riskScore = 0.0;

        // Location-based risk
        const locationRisk = await this.assessLocationRisk(
            context.sourceLocation,
            request.principal.identifier
        );
        riskScore += locationRisk * 0.2;

        // Device-based risk
        const deviceRisk = await this.assessDeviceRisk(context.deviceFingerprint);
        riskScore += deviceRisk * 0.2;

        // Behavioral risk
        const behaviorRisk = await this.assessBehavioralRisk(
            request.principal.identifier,
            request.resourceId,
            context.timestamp
        );
        riskScore += behaviorRisk * 0.3;

        // Trust level inverse correlation
        const trustRisk = (100 - trustLevel) / 100;
        riskScore += trustRisk * 0.2;

        // Resource sensitivity
        const resourceRisk = await this.assessResourceSensitivity(request.resourceId);
        riskScore += resourceRisk * 0.1;

        return Math.min(riskScore, 1.0); // Cap at 1.0
    }

    private async assessBehavioralRisk(
        principalId: string,
        resourceId: string,
        timestamp: Date
    ): Promise<number> {
        const historicalAccess = await this.getHistoricalAccess(principalId, resourceId);
        
        // Unusual time access
        const timeRisk = this.calculateTimeBasedRisk(timestamp, historicalAccess);
        
        // Frequency anomaly
        const frequencyRisk = this.calculateFrequencyRisk(historicalAccess);
        
        // Pattern deviation
        const patternRisk = await this.calculatePatternDeviationRisk(
            principalId,
            resourceId,
            historicalAccess
        );
        
        return (timeRisk + frequencyRisk + patternRisk) / 3;
    }
}
```

### CRDT Implementation for Distributed Data

**OR-Set (Observed-Remove Set) Implementation**:

```typescript
interface ORSetElement<T> {
    element: T;
    added: Set<string>; // Node IDs that added this element
    removed: Set<string>; // Node IDs that removed this element
}

class ORSet<T> {
    private elements: Map<string, ORSetElement<T>>;
    private nodeId: string;

    constructor(nodeId: string) {
        this.nodeId = nodeId;
        this.elements = new Map();
    }

    add(element: T): void {
        const elementKey = this.getElementKey(element);
        const existing = this.elements.get(elementKey);

        if (existing) {
            existing.added.add(this.nodeId);
        } else {
            this.elements.set(elementKey, {
                element,
                added: new Set([this.nodeId]),
                removed: new Set()
            });
        }
    }

    remove(element: T): void {
        const elementKey = this.getElementKey(element);
        const existing = this.elements.get(elementKey);

        if (existing && existing.added.size > 0) {
            existing.removed.add(this.nodeId);
        }
    }

    contains(element: T): boolean {
        const elementKey = this.getElementKey(element);
        const existing = this.elements.get(elementKey);

        if (!existing) return false;

        // Element is present if it was added by at least one node
        // and not removed by all nodes that added it
        const addedNodes = Array.from(existing.added);
        const removedNodes = Array.from(existing.removed);

        return addedNodes.some(nodeId => !removedNodes.includes(nodeId));
    }

    merge(other: ORSet<T>): void {
        for (const [elementKey, otherElement] of other.elements) {
            const existing = this.elements.get(elementKey);

            if (existing) {
                // Merge added and removed sets
                for (const nodeId of otherElement.added) {
                    existing.added.add(nodeId);
                }
                for (const nodeId of otherElement.removed) {
                    existing.removed.add(nodeId);
                }
            } else {
                // Create new element with copies of the sets
                this.elements.set(elementKey, {
                    element: otherElement.element,
                    added: new Set(otherElement.added),
                    removed: new Set(otherElement.removed)
                });
            }
        }
    }

    toArray(): T[] {
        const result: T[] = [];
        for (const [_, elementData] of this.elements) {
            if (this.contains(elementData.element)) {
                result.push(elementData.element);
            }
        }
        return result;
    }

    private getElementKey(element: T): string {
        return JSON.stringify(element);
    }
}
```

```python
    def add_node(self, node: NodeInfo) -> bool:
        """Add a node to the bucket. Returns True if added, False if bucket is full."""
        # Remove if already exists
        self.nodes = [n for n in self.nodes if n.node_id != node.node_id]
        
        if len(self.nodes) < self.max_size:
            self.nodes.append(node)
            return True
        else:
            # Bucket is full, implement replacement strategy
            # In Kademlia, we ping the least recently seen node
            return self._try_replace_node(node)
    
    def _try_replace_node(self, new_node: NodeInfo) -> bool:
        # For simplicity, replace the oldest node
        # In production, ping the oldest and replace if it doesn't respond
        if self.nodes:
            self.nodes[0] = new_node
            return True
        return False
    
    def find_closest_nodes(self, target_id: bytes, count: int) -> List[NodeInfo]:
        """Find the closest nodes to target_id by XOR distance."""
        def xor_distance(node_id: bytes) -> int:
            return int.from_bytes(
                bytes(a ^ b for a, b in zip(target_id, node_id)), 
                'big'
            )
        
        sorted_nodes = sorted(self.nodes, key=lambda n: xor_distance(n.node_id))
        return sorted_nodes[:count]

class DistributedHashTable:
    def __init__(self, node_id: bytes, ip_address: str, port: int):
        self.node_id = node_id
        self.node_info = NodeInfo(node_id, ip_address, port, time.time())
        self.buckets: List[KBucket] = [KBucket() for _ in range(160)]  # 160 bits for SHA-1
        self.storage: Dict[bytes, any] = {}
        self.pending_requests: Dict[str, asyncio.Future] = {}
        
    def _get_bucket_index(self, node_id: bytes) -> int:
        """Get the bucket index for a given node ID."""
        distance = int.from_bytes(
            bytes(a ^ b for a, b in zip(self.node_id, node_id)), 
            'big'
        )
        return distance.bit_length() - 1 if distance > 0 else 0
    
    async def store(self, key: bytes, value: any) -> bool:
        """Store a key-value pair in the DHT."""
        # Store locally
        self.storage[key] = value
        
        # Find closest nodes to the key
        closest_nodes = await self.find_closest_nodes(key, 20)
        
        # Store on closest nodes
        store_tasks = []
        for node in closest_nodes:
            if node.node_id != self.node_id:
                task = self._send_store_message(node, key, value)
                store_tasks.append(task)
        
        # Wait for majority of stores to complete
        results = await asyncio.gather(*store_tasks, return_exceptions=True)
        successful_stores = sum(1 for r in results if r is True)
        
        return successful_stores >= len(store_tasks) // 2
    
    async def find_value(self, key: bytes) -> Optional[any]:
        """Find a value in the DHT."""
        # Check local storage first
        if key in self.storage:
            return self.storage[key]
        
        # Search the network
        visited_nodes = set()
        candidate_nodes = await self.find_closest_nodes(key, 3)
        
        while candidate_nodes:
            # Sort by distance to key
            candidate_nodes.sort(
                key=lambda n: self._xor_distance(key, n.node_id)
            )
            
            current_node = candidate_nodes.pop(0)
            if current_node.node_id in visited_nodes:
                continue
            
            visited_nodes.add(current_node.node_id)
            
            # Query the node
            response = await self._send_find_value_message(current_node, key)
            
            if response and 'value' in response:
                return response['value']
            elif response and 'nodes' in response:
                # Add new nodes to candidates
                new_nodes = [
                    NodeInfo(node_data['id'], node_data['ip'], node_data['port'], time.time())
                    for node_data in response['nodes']
                    if node_data['id'] not in visited_nodes
                ]
                candidate_nodes.extend(new_nodes)
        
        return None
    
    async def find_closest_nodes(self, target_id: bytes, count: int) -> List[NodeInfo]:
        """Find the closest nodes to a target ID."""
        all_nodes = []
        
        # Collect nodes from all buckets
        for bucket in self.buckets:
            all_nodes.extend(bucket.nodes)
        
        # Sort by XOR distance to target
        all_nodes.sort(key=lambda n: self._xor_distance(target_id, n.node_id))
        
        return all_nodes[:count]
    
    def _xor_distance(self, id1: bytes, id2: bytes) -> int:
        """Calculate XOR distance between two node IDs."""
        return int.from_bytes(
            bytes(a ^ b for a, b in zip(id1, id2)), 
            'big'
        )
    
    async def _send_store_message(self, node: NodeInfo, key: bytes, value: any) -> bool:
        """Send a store message to a node."""
        message = DHTMessage(
            message_type=MessageType.STORE,
            sender=self.node_info,
            transaction_id=self._generate_transaction_id(),
            payload={'key': key, 'value': value}
        )
        
        try:
            response = await self._send_message(node, message)
            return response.get('success', False)
        except Exception:
            return False
    
    async def _send_find_value_message(self, node: NodeInfo, key: bytes) -> Optional[Dict]:
        """Send a find_value message to a node."""
        message = DHTMessage(
            message_type=MessageType.FIND_VALUE,
            sender=self.node_info,
            transaction_id=self._generate_transaction_id(),
            payload={'key': key}
        )
        
        try:
            return await self._send_message(node, message)
        except Exception:
            return None
    
    def add_node(self, node: NodeInfo):
        """Add a node to the routing table."""
        bucket_index = self._get_bucket_index(node.node_id)
        self.buckets[bucket_index].add_node(node)
    
    def _generate_transaction_id(self) -> str:
        """Generate a unique transaction ID."""
        return hashlib.sha1(
            f"{time.time()}{self.node_id}".encode()
        ).hexdigest()[:8]
```

```go
package consensus

import (
    "crypto/sha256"
    "encoding/binary"
    "errors"
    "math/big"
    "time"
)

type ContributionType int

const (
    NetworkContribution ContributionType = iota
    DevelopmentContribution
    CommunityContribution
    ServiceContribution
)

type ContributionRecord struct {
    ValidatorID      []byte
    ContributionType ContributionType
    Amount          *big.Int
    Timestamp       time.Time
    Evidence        []byte // Cryptographic proof of contribution
    Signature       []byte
}

type ProofOfContribution struct {
    validators       map[string]*Validator
    contributionLog  []*ContributionRecord
    currentEpoch     uint64
    epochDuration    time.Duration
    minValidators    int
}

type Validator struct {
    ID                []byte
    PublicKey         []byte
    StakeAmount       *big.Int
    ContributionScore *ContributionScore
    LastActive        time.Time
    ReputationScore   float64
}

type ContributionScore struct {
    NetworkScore     *big.Int
    DevelopmentScore *big.Int
    CommunityScore   *big.Int
    ServiceScore     *big.Int
    TotalScore       *big.Int
    LastUpdated      time.Time
}

func NewProofOfContribution(epochDuration time.Duration, minValidators int) *ProofOfContribution {
    return &ProofOfContribution{
        validators:      make(map[string]*Validator),
        contributionLog: make([]*ContributionRecord, 0),
        epochDuration:   epochDuration,
        minValidators:   minValidators,
        currentEpoch:    0,
    }
}

func (poc *ProofOfContribution) SelectValidators(count int) ([]*Validator, error) {
    if len(poc.validators) < poc.minValidators {
        return nil, errors.New("insufficient validators")
    }
    
    // Calculate selection probabilities based on contribution scores
    totalScore := big.NewInt(0)
    validatorList := make([]*Validator, 0, len(poc.validators))
    
    for _, validator := range poc.validators {
        if poc.isValidatorActive(validator) {
            totalScore.Add(totalScore, validator.ContributionScore.TotalScore)
            validatorList.append(validator)
        }
    }
    
    if totalScore.Cmp(big.NewInt(0)) == 0 {
        return nil, errors.New("no active validators with contributions")
    }
    
    // Weighted random selection
    selected := make([]*Validator, 0, count)
    for i := 0; i < count && len(validatorList) > 0; i++ {
        selectedValidator := poc.weightedRandomSelect(validatorList, totalScore)
        selected = append(selected, selectedValidator)
        
        // Remove selected validator from list to avoid duplicates
        validatorList = poc.removeValidator(validatorList, selectedValidator)
        totalScore.Sub(totalScore, selectedValidator.ContributionScore.TotalScore)
    }
    
    return selected, nil
}

func (poc *ProofOfContribution) ValidateBlock(block *Block, proposer *Validator) error {
    // Verify proposer eligibility
    if !poc.isValidatorEligible(proposer) {
        return errors.New("proposer not eligible")
    }
    
    // Verify block structure and transactions
    if err := poc.validateBlockStructure(block); err != nil {
        return err
    }
    
    // Verify proposer signature
    if err := poc.verifyProposerSignature(block, proposer); err != nil {
        return err
    }
    
    // Verify contribution proofs in block
    for _, contributionProof := range block.ContributionProofs {
        if err := poc.validateContributionProof(contributionProof); err != nil {
            return err
        }
    }
    
    return nil
}

func (poc *ProofOfContribution) UpdateContributionScore(
    validatorID []byte,
    contributionType ContributionType,
    evidence []byte,
) error {
    validatorKey := string(validatorID)
    validator, exists := poc.validators[validatorKey]
    if !exists {
        return errors.New("validator not found")
    }
    
    // Calculate contribution amount based on evidence
    contributionAmount, err := poc.calculateContributionAmount(contributionType, evidence)
    if err != nil {
        return err
    }
    
    // Update validator's contribution score
    switch contributionType {
    case NetworkContribution:
        validator.ContributionScore.NetworkScore.Add(
            validator.ContributionScore.NetworkScore,
            contributionAmount,
        )
    case DevelopmentContribution:
        validator.ContributionScore.DevelopmentScore.Add(
            validator.ContributionScore.DevelopmentScore,
            contributionAmount,
        )
    case CommunityContribution:
        validator.ContributionScore.CommunityScore.Add(
            validator.ContributionScore.CommunityScore,
            contributionAmount,
        )
    case ServiceContribution:
        validator.ContributionScore.ServiceScore.Add(
            validator.ContributionScore.ServiceScore,
            contributionAmount,
        )
    }
    
    // Recalculate total score with weighted average
    poc.recalculateTotalScore(validator)
    
    // Create contribution record
    record := &ContributionRecord{
        ValidatorID:      validatorID,
        ContributionType: contributionType,
        Amount:          contributionAmount,
        Timestamp:       time.Now(),
        Evidence:        evidence,
    }
    
    // Sign the record
    signature, err := poc.signContributionRecord(record)
    if err != nil {
        return err
    }
    record.Signature = signature
    
    // Add to contribution log
    poc.contributionLog = append(poc.contributionLog, record)
    
    return nil
}

func (poc *ProofOfContribution) recalculateTotalScore(validator *Validator) {
    // Weighted combination of different contribution types
    weights := map[ContributionType]float64{
        NetworkContribution:     0.3,
        DevelopmentContribution: 0.25,
        CommunityContribution:   0.25,
        ServiceContribution:     0.2,
    }
    
    totalScore := big.NewInt(0)
    
    // Apply weights and combine scores
    networkWeighted := new(big.Int).Mul(
        validator.ContributionScore.NetworkScore,
        big.NewInt(int64(weights[NetworkContribution] * 100)),
    )
    networkWeighted.Div(networkWeighted, big.NewInt(100))
    
    devWeighted := new(big.Int).Mul(
        validator.ContributionScore.DevelopmentScore,
        big.NewInt(int64(weights[DevelopmentContribution] * 100)),
    )
    devWeighted.Div(devWeighted, big.NewInt(100))
    
    communityWeighted := new(big.Int).Mul(
        validator.ContributionScore.CommunityScore,
        big.NewInt(int64(weights[CommunityContribution] * 100)),
    )
    communityWeighted.Div(communityWeighted, big.NewInt(100))
    
    serviceWeighted := new(big.Int).Mul(
        validator.ContributionScore.ServiceScore,
        big.NewInt(int64(weights[ServiceContribution] * 100)),
    )
    serviceWeighted.Div(serviceWeighted, big.NewInt(100))
    
    totalScore.Add(totalScore, networkWeighted)
    totalScore.Add(totalScore, devWeighted)
    totalScore.Add(totalScore, communityWeighted)
    totalScore.Add(totalScore, serviceWeighted)
    
    validator.ContributionScore.TotalScore = totalScore
    validator.ContributionScore.LastUpdated = time.Now()
}

func (poc *ProofOfContribution) calculateContributionAmount(
    contributionType ContributionType,
    evidence []byte,
) (*big.Int, error) {
    switch contributionType {
    case NetworkContribution:
        return poc.calculateNetworkContribution(evidence)
    case DevelopmentContribution:
        return poc.calculateDevelopmentContribution(evidence)
    case CommunityContribution:
        return poc.calculateCommunityContribution(evidence)
    case ServiceContribution:
        return poc.calculateServiceContribution(evidence)
    default:
        return nil, errors.New("unknown contribution type")
    }
}

func (poc *ProofOfContribution) calculateNetworkContribution(evidence []byte) (*big.Int, error) {
    // Parse evidence for network metrics
    // Evidence format: {bandwidth_provided, uptime_percentage, packets_forwarded}
    
    if len(evidence) < 24 { // 3 * 8 bytes
        return nil, errors.New("insufficient evidence data")
    }
    
    bandwidth := binary.BigEndian.Uint64(evidence[0:8])     // bytes/second
    uptime := binary.BigEndian.Uint64(evidence[8:16])       // percentage * 100
    packets := binary.BigEndian.Uint64(evidence[16:24])     // packet count
    
    // Calculate contribution score based on metrics
    // Formula: (bandwidth * uptime_factor * packets) / normalization_factor
    uptimeFactor := float64(uptime) / 10000.0 // Convert to 0-1 range
    
    contribution := float64(bandwidth) * uptimeFactor * float64(packets) / 1000000.0
    
    return big.NewInt(int64(contribution)), nil
}
```
